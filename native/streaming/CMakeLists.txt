cmake_minimum_required(VERSION 3.24)
project(SoftwareFuserStreaming LANGUAGES C CXX)
get_filename_component(FUSER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(FUSER_MOONLIGHT_SOURCE "${FUSER_ROOT}/build/references/moonlight-common-c" CACHE PATH "Pinned official Moonlight streaming core")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "WindowsStore")
  message(FATAL_ERROR "This target validates the UWP streaming libraries. Use WindowsStore.")
endif()
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
set(BUILD_SHARED_LIBS OFF)
find_package(OpenSSL REQUIRED)
find_package(CURL CONFIG REQUIRED)
find_package(expat CONFIG REQUIRED)
find_package(FFMPEG REQUIRED)
# The upstream core moves the source mouse when a connection starts. Build an
# isolated copy with that wake-up removed for this view-only widget.
set(FUSER_MOONLIGHT_BUILD_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/moonlight-source")
file(MAKE_DIRECTORY "${FUSER_MOONLIGHT_BUILD_SOURCE}")
file(COPY "${FUSER_MOONLIGHT_SOURCE}/CMakeLists.txt" "${FUSER_MOONLIGHT_SOURCE}/src"
  "${FUSER_MOONLIGHT_SOURCE}/enet" "${FUSER_MOONLIGHT_SOURCE}/nanors"
  DESTINATION "${FUSER_MOONLIGHT_BUILD_SOURCE}")
file(READ "${FUSER_MOONLIGHT_BUILD_SOURCE}/src/Connection.c" FUSER_CONNECTION_SOURCE)
set(FUSER_MOUSE_WAKEUP "    // Wiggle the mouse a bit to wake the display up\n    LiSendMouseMoveEvent(1, 1);\n    PltSleepMs(10);\n    LiSendMouseMoveEvent(-1, -1);\n    PltSleepMs(10);")
string(FIND "${FUSER_CONNECTION_SOURCE}" "${FUSER_MOUSE_WAKEUP}" FUSER_MOUSE_OFFSET)
if(FUSER_MOUSE_OFFSET LESS 0)
  message(FATAL_ERROR "Review upstream mouse wake-up changes before building the view-only core.")
endif()
string(REPLACE "${FUSER_MOUSE_WAKEUP}" "    // Software Fuser: view-only startup sends no mouse movement."
  FUSER_CONNECTION_SOURCE "${FUSER_CONNECTION_SOURCE}")
file(WRITE "${FUSER_MOONLIGHT_BUILD_SOURCE}/src/Connection.c" "${FUSER_CONNECTION_SOURCE}")
add_subdirectory("${FUSER_MOONLIGHT_BUILD_SOURCE}" moonlight)
add_library(fuser_decoder STATIC ../windows/FFmpegDecoder.cpp)
target_compile_features(fuser_decoder PUBLIC cxx_std_20)
target_compile_options(fuser_decoder PRIVATE /W4 /permissive-)
target_compile_definitions(fuser_decoder PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(fuser_decoder PUBLIC ../windows ../core/include ${FFMPEG_INCLUDE_DIRS})
target_link_directories(fuser_decoder PUBLIC ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(fuser_decoder PUBLIC ${FFMPEG_LIBRARIES} d3d11)
add_library(fuser_control STATIC SunshineControl.cpp)
target_compile_features(fuser_control PUBLIC cxx_std_20)
target_compile_options(fuser_control PRIVATE /W4 /permissive-)
target_compile_definitions(fuser_control PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(fuser_control PUBLIC . ../core/include)
target_link_libraries(fuser_control PUBLIC moonlight-common-c CURL::libcurl expat::expat OpenSSL::SSL OpenSSL::Crypto)
add_library(fuser_session STATIC OverlaySession.cpp)
target_compile_features(fuser_session PUBLIC cxx_std_20)
target_compile_options(fuser_session PRIVATE /W4 /permissive-)
target_compile_definitions(fuser_session PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(fuser_session PUBLIC . ../windows ../core/include)
target_link_libraries(fuser_session PUBLIC fuser_control fuser_decoder)
